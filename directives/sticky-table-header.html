# 表格头部固定指令

# 基础用法

使用 v-sticky-table-header 指令可以使表格头部固定。

<template>
  <div class="example-sticky-table">
    <el-table
      v-sticky-table-header
      :data="tableData"
      style="width: 100%"
    >
      <el-table-column
        prop="date"
        label="日期"
        width="180">
      </el-table-column>
      <el-table-column
        prop="name"
        label="姓名"
        width="180">
      </el-table-column>
      <el-table-column
        prop="address"
        label="地址">
      </el-table-column>
    </el-table>
  </div>
</template>

<script>
export default {
  data() {
    return {
      tableData: Array.from({ length: 10 }, (_, index) => ({
        date: `2024-01-${String(index + 1).padStart(2, '0')}`,
        name: `张三${index + 1}`,
        address: `浙江省杭州市西湖区某某街道${index + 1}号`
      }))
    }
  }
}
</script>

<style scoped>
.example-sticky-table {
  padding: 20px;
  height: 300px;
  overflow: auto;
}
</style>
