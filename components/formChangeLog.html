# FormChangeLog 表单变更记录

用于展示表单数据的变更记录，支持多种数据类型的对比展示。

# 基础用法

展示嵌套对象的变更记录，支持平铺展示。

<template>
  <vin-form-change-log
    :old-data="{
      address: {
        province: '北京市',
        city: '朝阳区',
        detail: '霞光里8号2号楼一层8128'
      }
    }"
    :new-data="{
      address: {
        province: '北京市',
        city: '顺义区',
        detail: '空港街道安泰大街九号院20号楼109-357'
      }
    }"
    :field-config="{
      address: {
        label: '地址信息',
        time: '2022-02-22',
        flatten: true,
        nestedLabels: {
          province: '省/市',
          city: '区/县',
          detail: '详细地址'
        }
      }
    }"
    :show-delete-line="false"
    :before-color="'#f38b00'"
    :after-color="'#0095d3'"
  />
</template>

# 数组类型

展示数组类型的变更记录，支持简单数组和下拉框数组。

<template>
  <div class="change-log-demo">
    <vin-form-change-log
      :old-data="oldData"
      :new-data="newData"
      :field-config="fieldConfig"
      :show-delete-line="false"
      :before-color="'#f38b00'"
      :after-color="'#0095d3'"
    />
  </div>
</template>

<script>
export default {
  name: 'ArrayFormChangeLog',
  data() {
    return {
      oldData: {
        hobbies: ['读书', '音乐'],
        tags: [1, 2]
      },
      newData: {
        hobbies: ['读书', '运动'],
        tags: [1, 3]
      },
      fieldConfig: {
        hobbies: {
          label: '爱好',
          type: 'array'
        },
        tags: {
          label: '标签',
          type: 'array',
          enumMap: {
            1: '前端',
            2: '后端',
            3: '设计'
          }
        }
      }
    };
  }
};
</script>

# 枚举类型

展示枚举类型的变更记录，支持值映射显示。

<template>
  <div class="change-log-demo">
    <vin-form-change-log
      :old-data="oldData"
      :new-data="newData"
      :field-config="fieldConfig"
      :show-delete-line="false"
      :before-color="'#f38b00'"
      :after-color="'#0095d3'"
    />
  </div>
</template>

<script>
export default {
  name: 'EnumFormChangeLog',
  data() {
    return {
      oldData: {
        gender: 1,
        status: 'ACTIVE',
        type: 'USER'
      },
      newData: {
        gender: 2,
        status: 'INACTIVE',
        type: 'ADMIN'
      },
      fieldConfig: {
        gender: {
          label: '性别',
          enumMap: {
            1: '男',
            2: '女'
          }
        },
        status: {
          label: '状态',
          enumMap: {
            'ACTIVE': '活跃',
            'INACTIVE': '停用'
          }
        },
        type: {
          label: '类型',
          enumMap: {
            'USER': '普通用户',
            'ADMIN': '管理员'
          }
        }
      }
    };
  }
};
</script>

# 对象数组比较

展示对象数组的变更记录，支持基于特定键的比较，并高亮显示字段变更。可以使用姓名等字段作为标题，而不是显示主键。

<template>
  <div class="change-log-demo">
    <vin-form-change-log
      :old-data="{
        members: [
          { id: 1, name: '张三', role: 'admin', department: '技术部' },
          { id: 2, name: '李四', role: 'user', department: '市场部' },
          { id: 3, name: '王五', role: 'user', department: '销售部' },
          { id: 5, name: '芦一', role: 'user', department: '销售部' }
        ]
      }"
      :new-data="{
        members: [
          { id: 1, name: '张三', role: 'user', department: '技术部' },
          { id: 3, name: '王五', role: 'admin', department: '技术部' },
          { id: 4, name: '赵六', role: 'user', department: '人事部' },
          { id: 5, name: '芦一', role: 'user', department: '销售部' }
        ]
      }"
      :field-config="{
        members: {
          label: '团队成员变更',
          type: 'objectArray',
          compareKey: 'id',
          titleField: 'name',
          showUnchanged: true,
          excludeFields: ['id'],
          time: '2023-05-18',
          fieldLabels: {
            name: '姓名',
            role: '角色',
            department: '部门'
          },
          fieldEnumMaps: {
            role: {
              'admin': '管理员',
              'user': '普通用户'
            }
          }
        }
      }"
      :show-delete-line="false"
      :before-color="'#f38b00'"
      :after-color="'#0095d3'"
    />
  </div>
</template>

<script>
export default {
  name: 'ObjectArrayFormChangeLog'
};
</script>

  padding: 20px;

# 属性

参数 说明 类型 默认值
oldData 变更前的数据 Object {}
newData 变更后的数据 Object {}
fieldConfig 字段配置 Object {}
showNullChanges 是否显示空值变更 Boolean false
showDeleteLine 是否显示删除线 Boolean true
beforeColor 变更前的文字颜色 String '#909399'
afterColor 变更后的文字颜色 String '#67c23a'

# fieldConfig 配置项

参数 说明 类型 默认值
label 字段标签 String -
time 变更时间 String -
type 字段类型，支持 'array'(普通数组)、'select'(下拉框数组)、'objectArray'(对象数组) String -
enumMap 枚举映射 Object -
flatten 是否平铺显示嵌套对象 Boolean false
nestedLabels 嵌套字段的标签映射 Object -
nestedEnumMaps 嵌套字段的枚举映射 Object -
compareKey 对象数组比较的键 String -
titleField 对象数组项的标题字段 String -
showUnchanged 是否显示未变更的对象 Boolean true
visibleFields 需要显示的字段列表 Array -
excludeFields 需要排除的字段列表 Array ['id']
fieldLabels 对象数组中字段的标签映射 Object -
fieldEnumMaps 对象数组中字段的枚举映射 Object -
